                openModal(item, market) {
                    this.activeItem = item;
                    this.activeMarket = market;
                    this.modalOpen = true;
                },

                // 为市场视图提取高光点 (展示该市场最严格或有差异的条目)
                getMarketHighlights(marketCode) {
                    return this.items
                        .filter(item => {
                            const s = item.details[marketCode]?.status;
                            // 展示不满足或开放点/需测试的条目
                            return s === 'not_fullfill' || s === 'open_need_test';
                        })
                        .map(item => ({
                            itemId: item.id,
                            text: this.getLangText(item.details[marketCode].summary) + '。' + 
      this.getLangText(item.details[marketCode].full).substring(0, 40) + '...'
                        }));
                },

                init(){
    const saved = localStorage.getItem('autoRegData')
    if(saved){
        const data = JSON.parse(saved)

        // 🔥 修复 items
        data.items.forEach(item=>{
            if(typeof item.title === 'string'){
                item.title = { zh:item.title, en:item.title }
            }

            if(typeof item.globalNote === 'string'){
    item.globalNote = { zh:item.globalNote, en:item.globalNote }
}

Object.values(item.details).forEach(d=>{
    ['summary','full','special','test'].forEach(k=>{
        if(typeof d[k] === 'string'){
            d[k] = { zh:d[k], en:d[k] }
        }
    })
})
        })

        // 🔥 修复 markets
        data.markets.forEach(m=>{
            if(typeof m.name === 'string'){
                m.name = { zh:m.name, en:m.name }
            }
            if(typeof m.standard === 'string'){
                m.standard = { zh:m.standard, en:m.standard }
            }
            if(typeof m.description === 'string'){
                m.description = { zh:m.description, en:m.description }
            }
        })

        this.items = data.items
        this.markets = data.markets
        if(data.groups) this.groups = data.groups
    }

    this.$watch('items', () => this.save(), { deep: true })
    this.$watch('markets', () => this.save(), { deep: true })
},

save(){
    clearTimeout(this._saveTimer)
    this._saveTimer = setTimeout(()=>{
        localStorage.setItem('autoRegData', JSON.stringify({
            items:this.items,
            markets:this.markets,
    groups:this.groups
        }))
    }, 500)
},

addItem(){
    const id = Date.now()
    const newItem = {
        id,
        groupId:'interior',
        regCode:'NEW',
        title:{ zh:'新条目', en:'New Item' },
        globalNote:{ zh:'', en:'' },
        details:{}
    }

    this.markets.forEach(m=>{
        newItem.details[m.code]={
            status:'fullfill',
            summary:{ zh:'', en:'' },
            full:{ zh:'', en:'' },
            special:{ zh:'', en:'' },
            test:{ zh:'', en:'' }
        }
    })

    this.items.push(newItem)
},

addMarket(){
    const code='M'+Date.now()

    const newMarket={
        code,
        name:{ zh:'新国家', en:'New Market' },
        standard:{ zh:'NEW', en:'NEW' },
        description:{ zh:'', en:'' }
    }

    this.markets.push(newMarket)

    this.items.forEach(item=>{
        item.details[code]={
            status:'fullfill',
            summary:{ zh:'', en:'' },
            full:{ zh:'', en:'' },
            special:{ zh:'', en:'' },
            test:{ zh:'', en:'' }
        }
    })
},

deleteItem(id){
    this.items = this.items.filter(item => item.id !== id)
},

deleteMarket(code){
    this.markets = this.markets.filter(m => m.code !== code)

    this.items = this.items.map(item => {
        const newDetails = { ...item.details }
        delete newDetails[code]
        return { ...item, details: newDetails }
    })
}
